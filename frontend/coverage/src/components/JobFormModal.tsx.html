import { createJob, updateJob } from '../api'
import { ALL_STATUSES, STATUS_LABEL, type ApplicationStatus, type JobPosting } from '../types'
 
interface JobFormModalProps {

  onSaved: () => void
}
 
// 공고 추가/수정 겸용 모달
